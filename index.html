#!/usr/bin/env node /** * export-node.js * Renders the church scene frame-by-frame via Puppeteer, * then encodes to H.264 MP4 with FFmpeg. * * Usage: * 1. npm install puppeteer (one time) * 2. Serve the HTML locally: * npx http-server . -p 8080 * 3. In another terminal: * node export-node.js * * Requires ffmpeg on PATH. */ const puppeteer = require('puppeteer'); const { spawn } = require('child_process'); const path = require('path'); const URL = process.env.SCENE_URL || 'http://localhost:8080/index.html'; const FPS = 60; const DURATION = 15; // seconds const TOTAL = FPS * DURATION; // 900 frames const OUT_W = 1920; const OUT_H = 1080; const OUTPUT = process.env.OUTPUT || 'church-loop.mp4'; (async () => { console.log(`Launching browser…`); const browser = await puppeteer.launch({ headless: 'new', args: ['--no-sandbox', '--disable-setuid-sandbox'] }); const page = await browser.newPage(); await page.setViewport({ width: OUT_W, height: OUT_H, deviceScaleFactor: 1 }); console.log(`Loading ${URL}…`); await page.goto(URL, { waitUntil: 'networkidle0', timeout: 60000 }); // Hide panel and force scene to fill viewport await page.evaluate(() => { document.getElementById('panel').style.display = 'none'; document.documentElement.style.setProperty('--panel-h', '0px'); window.dispatchEvent(new Event('resize')); }); // Wait for PNG overlay to load await page.waitForFunction( () => { const img = document.getElementById('baseLayer'); return img && img.complete && img.naturalWidth > 0; }, { timeout: 30000 } ); // Let layout settle await new Promise(r => setTimeout(r, 1000)); console.log(`Starting FFmpeg…`); const ffmpeg = spawn('ffmpeg', [ '-y', '-f', 'image2pipe', '-framerate', String(FPS), '-i', '-', '-c:v', 'libx264', '-pix_fmt', 'yuv420p', '-preset', 'slow', '-crf', '18', '-r', String(FPS), '-movflags', '+faststart', OUTPUT ], { stdio: ['pipe', 'inherit', 'inherit'] }); ffmpeg.on('error', e => { console.error('FFmpeg error:', e); process.exit(1); }); console.log(`Rendering ${TOTAL} frames…`); for (let i = 0; i < TOTAL; i++) { // Ask the page to render this frame and return PNG base64 const b64 = await page.evaluate((fi, total) => { return window.renderExportFrameAt(fi, total); }, i, TOTAL); // Strip data URL header → raw base64 → Buffer const buf = Buffer.from(b64.replace(/^data:image\/png;base64,/, ''), 'base64'); ffmpeg.stdin.write(buf); if (i % 60 === 0 || i === TOTAL - 1) { console.log(` frame ${i + 1} / ${TOTAL} (${Math.round((i + 1) / TOTAL * 100)}%)`); } } ffmpeg.stdin.end(); await new Promise(resolve => ffmpeg.on('close', resolve)); await browser.close(); console.log(`\n✅ Done → ${path.resolve(OUTPUT)}`); })();
